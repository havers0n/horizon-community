```javascript
// Исправленная функция sendToClient - отключен invokeNative
function sendToClient(data) {
    try {
        // Метод 1: fetch (самый надежный способ)
        fetch('https://mdt-system/nuiCallback', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(data)
        });
        
        // Метод 2: $.post (jQuery, если доступен)
        if (typeof $ !== 'undefined') {
            $.post('https://mdt-system/nuiCallback', JSON.stringify(data));
        }
        
        // invokeNative отключен - не работает в данном контексте
    } catch (error) {
        console.error('[MDT Test] Ошибка отправки:', error);
    }
}
```
