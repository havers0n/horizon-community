```lua
-- Универсальная проверка разрешений
local function hasPermission(source)
    local playerData = exports[GetCurrentResourceName()]:MDTGetPlayerData(source)
    if not playerData or not playerData.job then
        return false
    end
    
    return exports[GetCurrentResourceName()]:MDTHasPermission(source, playerData.job.name)
end

-- Rate limiting для защиты от спама
if not exports[GetCurrentResourceName()]:MDTRateLimitCheck(source) then
    notify(source, "Слишком много запросов. Попробуйте позже.", "warning")
    return
end
```
